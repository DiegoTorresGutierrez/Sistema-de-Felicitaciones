<script>
  let APP = {
    people: [],
    categories: [],
    selectedCategoryId: null
  };

  document.addEventListener('DOMContentLoaded', loadApp);

  function loadApp() {
    google.script.run
      .withSuccessHandler(initializeApp)
      .withFailureHandler(showError)
      .getAppData();
  }

  function initializeApp(data) {
    APP.people = data.people || [];
    APP.categories = data.categories || [];
    
    renderTops(data.stats);
    renderFilters(data.departments);
    populateModalSelect(APP.people);
    renderModalCategories();
    
    // Al iniciar, filtramos e imprimimos el grid
    filterGrid(); 
  }

  /* RENDERIZADO DE TOPS */
  function renderTops(stats) {
    document.getElementById('statMonth').textContent = stats.thisMonth;
    document.getElementById('statRecognized').textContent = stats.peopleRecognized;
    document.getElementById('statTotal').textContent = stats.total;

    // Top Personas (General y Mes)
    const topPeopleHtml = stats.topPeople.map(p => `
      <div class="top-item">
        <div class="top-item-info">
          <img src="${escapeHtml(p.photoUrl)}" class="top-avatar" onerror="this.src='https://via.placeholder.com/300x300.png?text=Emp'">
          <div>
            <div style="font-weight: bold; font-size: 14px; color: var(--secondary)">${escapeHtml(p.name)}</div>
            <div style="font-size: 11px; color: var(--muted)">${escapeHtml(p.department)}</div>
          </div>
        </div>
        <div class="top-stats">
          Total: ${p.recognitions}<br>
          <span style="font-size: 10px; color: var(--muted);">Mes: ${p.monthRecognitions}</span>
        </div>
      </div>
    `).join('');
    document.getElementById('topPeopleContainer').innerHTML = topPeopleHtml || '<p>No hay datos.</p>';

    // Top Departamentos
    const topDeptHtml = stats.topDepartments.map(d => `
      <div class="top-item">
        <div style="font-weight: bold; color: var(--secondary)">🏢 ${escapeHtml(d.department)}</div>
        <div class="top-stats">🏆 ${d.count}</div>
      </div>
    `).join('');
    document.getElementById('topDeptContainer').innerHTML = topDeptHtml || '<p>No hay datos.</p>';
  }

  /* RENDERIZADO DE FILTROS Y GRID PRINCIPAL */
  function renderFilters(departments) {
    const select = document.getElementById('deptFilter');
    departments.forEach(dept => {
      const opt = document.createElement('option');
      opt.value = dept; opt.textContent = dept;
      select.appendChild(opt);
    });
  }

  function filterGrid() {
    const query = document.getElementById('searchInput').value.toLowerCase().trim();
    const deptFilter = document.getElementById('deptFilter').value;

    const filtered = APP.people.filter(person => {
      // REGLA CRÍTICA: Solo personal con 1 o más felicitaciones
      if (person.recognitions === 0) return false;
      
      const matchName = String(person.name).toLowerCase().includes(query);
      const matchDept = deptFilter === "" || person.department === deptFilter;
      return matchName && matchDept;
    });

    renderMainGrid(filtered);
  }

  function renderMainGrid(peopleList) {
    const container = document.getElementById('peopleContainer');
    if (peopleList.length === 0) {
      container.innerHTML = '<div style="grid-column: 1/-1; text-align: center; color: var(--muted);">No se encontraron reconocimientos bajo estos filtros.</div>';
      return;
    }

    container.innerHTML = peopleList.map(person => `
      <div class="person-card">
        <img class="person-photo" src="${escapeHtml(person.photoUrl)}" onerror="this.src='https://via.placeholder.com/300x300.png?text=Usuario'">
        <div class="person-name">${escapeHtml(person.name)}</div>
        <div class="person-department">${escapeHtml(person.department)}</div>
        <div class="person-count">🏆 ${person.recognitions} Reconocimientos</div>
      </div>
    `).join('');
  }

  /* MODAL DE NUEVA FELICITACIÓN */
  function openNewRecognitionModal() {
    document.getElementById('modal').style.display = 'flex';
    document.getElementById('message').value = '';
    APP.selectedCategoryId = null;
    document.querySelectorAll('.category').forEach(el => el.classList.remove('selected'));
    document.getElementById('personSelect').value = "";
  }
  function closeModal() {
    document.getElementById('modal').style.display = 'none';
  }

  // Populate list with ALL active users (Incluyendo 0 reconocimientos)
  function populateModalSelect(people) {
    const select = document.getElementById('personSelect');
    // Ordenar alfabéticamente para mayor comodidad
    const sorted = [...people].sort((a,b) => a.name.localeCompare(b.name));
    
    sorted.forEach(p => {
      const opt = document.createElement('option');
      opt.value = p.id; 
      opt.textContent = `${p.name} (${p.department})`;
      select.appendChild(opt);
    });
  }

  function renderModalCategories() {
    const container = document.getElementById('categoriesContainer');
    container.innerHTML = APP.categories.map(c => `
      <div class="category" data-id="${escapeHtml(c.id)}" onclick="selectCategory('${escapeJs(c.id)}')">
        <span class="category-emoji">${escapeHtml(c.emoji)}</span>
        <span class="category-name">${escapeHtml(c.name)}</span>
      </div>
    `).join('');
  }

  function selectCategory(id) {
    APP.selectedCategoryId = id;
    document.querySelectorAll('.category').forEach(el => {
      el.classList.toggle('selected', el.dataset.id === String(id));
    });
  }

  /* ENVÍO DE DATOS */
  function sendRecognition() {
    const personId = document.getElementById('personSelect').value;
    const msg = document.getElementById('message').value.trim();

    if (!personId) return showToast('Por favor, selecciona a quién vas a felicitar.');
    if (!APP.selectedCategoryId) return showToast('Por favor, selecciona una categoría.');
    if (!msg) return showToast('Escribe un mensaje de reconocimiento.');

    document.getElementById('submitBtn').textContent = "Enviando...";
    document.getElementById('submitBtn').disabled = true;

    google.script.run
      .withSuccessHandler(res => {
        closeModal();
        showToast(`🎉 ¡Reconocimiento enviado a ${res.personName}!`);
        document.getElementById('submitBtn').textContent = "Enviar Reconocimiento";
        document.getElementById('submitBtn').disabled = false;
        loadApp(); // Recargar datos para mostrar los cambios
      })
      .withFailureHandler(err => {
        showError(err);
        document.getElementById('submitBtn').textContent = "Enviar Reconocimiento";
        document.getElementById('submitBtn').disabled = false;
      })
      .createRecognition({
        personId: personId,
        categoryId: APP.selectedCategoryId,
        message: msg
      });
  }

  /* UTILIDADES UI */
  function showToast(msg) {
    const toast = document.getElementById('toast');
    toast.textContent = msg;
    toast.style.display = 'block';
    setTimeout(() => toast.style.display = 'none', 4000);
  }

  function showError(err) {
    console.error(err);
    showToast(err.message || 'Ocurrió un error inesperado.');
  }

  function escapeHtml(value) {
    return String(value ?? '').replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;').replace(/"/g, '&quot;').replace(/'/g, '&#039;');
  }

  function escapeJs(value) {
    return String(value ?? '').replace(/\\/g, '\\\\').replace(/'/g, "\\'");
  }
</script>
