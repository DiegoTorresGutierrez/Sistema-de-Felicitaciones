<script>
  let APP_STATE = {
    people: [],
    categories: [],
    selectedPersonId: null,
    selectedCategoryId: null
  };

  document.addEventListener('DOMContentLoaded', () => {
    google.script.run
      .withSuccessHandler(initApp)
      .withFailureHandler(handleError)
      .getAppData();
  });

  function initApp(data) {
    document.getElementById('loadingScreen').style.display = 'none';
    document.getElementById('mainContent').style.display = 'block';
    
    document.getElementById('userProfile').innerText = 'Hola, ' + data.currentUser.name;

    APP_STATE.people = data.people;
    APP_STATE.categories = data.categories;

    renderDashboard(data.dashboard);
    populateDeptFilter();
    renderPeople(APP_STATE.people);
  }

  function renderDashboard(dashData) {
    // Top Personas (Podio)
    const podium = document.getElementById('podiumContainer');
    if (dashData.topPeople.length === 0) {
      podium.innerHTML = '<p style="color:var(--muted); width:100%; text-align:center;">Aún no hay reconocimientos.</p>';
    } else {
      // Reordenar para visual (2, 1, 3)
      const visualOrder = [];
      if(dashData.topPeople[1]) visualOrder.push({...dashData.topPeople[1], rank: 2});
      if(dashData.topPeople[0]) visualOrder.push({...dashData.topPeople[0], rank: 1});
      if(dashData.topPeople[2]) visualOrder.push({...dashData.topPeople[2], rank: 3});

      podium.innerHTML = visualOrder.map(p => `
        <div class="podium-item rank-${p.rank}">
          <img src="${escapeHtml(p.photoUrl)}" alt="${escapeHtml(p.name)}">
          <span class="podium-name">${escapeHtml(p.name)}</span>
          <div class="rank-bar">${p.recognitions}</div>
        </div>
      `).join('');
    }

    // Top Deptos
    const deptList = document.getElementById('deptContainer');
    deptList.innerHTML = dashData.topDepartments.map(d => `
      <li class="dept-item">
        <span>${escapeHtml(d.department)}</span>
        <span class="dept-count">${d.count}</span>
      </li>
    `).join('');

    // Stats
    document.getElementById('statTotal').innerText = dashData.stats.total;
    document.getElementById('statMonth').innerText = dashData.stats.thisMonth;
  }

  function populateDeptFilter() {
    const depts = [...new Set(APP_STATE.people.map(p => p.department))].filter(Boolean).sort();
    const select = document.getElementById('deptFilter');
    depts.forEach(d => {
      const opt = document.createElement('option');
      opt.value = d;
      opt.textContent = d;
      select.appendChild(opt);
    });
  }

  function renderPeople(list) {
    const container = document.getElementById('peopleContainer');
    container.innerHTML = list.map(p => `
      <div class="person-card" onclick="openModal(${p.id})">
        <img class="person-photo" src="${escapeHtml(p.photoUrl)}" alt="${escapeHtml(p.name)}">
        <div class="person-name">${escapeHtml(p.name)}</div>
        <div class="person-dept">${escapeHtml(p.department)}</div>
        <button class="btn-felicitar">Felicitar</button>
      </div>
    `).join('');
  }

  function filterPeople() {
    const q = document.getElementById('searchInput').value.toLowerCase();
    const d = document.getElementById('deptFilter').value;
    
    const filtered = APP_STATE.people.filter(p => {
      const matchName = p.name.toLowerCase().includes(q);
      const matchDept = d === '' || p.department === d;
      return matchName && matchDept;
    });
    renderPeople(filtered);
  }

  function openModal(personId) {
    const person = APP_STATE.people.find(p => p.id === personId);
    if(!person) return;

    APP_STATE.selectedPersonId = personId;
    APP_STATE.selectedCategoryId = null;

    document.getElementById('modalTarget').innerHTML = `
      <img src="${escapeHtml(person.photoUrl)}" alt="Avatar">
      <div>
        <h3>Felicitar a ${escapeHtml(person.name)}</h3>
        <p>${escapeHtml(person.department)}</p>
      </div>
    `;

    const catContainer = document.getElementById('modalCategories');
    catContainer.innerHTML = APP_STATE.categories.map(c => `
      <div class="cat-item" id="cat-${c.id}" onclick="selectCategory(${c.id})">
        <span class="cat-emoji">${c.emoji}</span>
        <span class="cat-name">${escapeHtml(c.name)}</span>
      </div>
    `).join('');

    document.getElementById('messageInput').value = '';
    document.getElementById('btnSubmit').disabled = false;
    document.getElementById('modal').style.display = 'flex';
  }

  function closeModal() {
    document.getElementById('modal').style.display = 'none';
  }

  function selectCategory(catId) {
    APP_STATE.selectedCategoryId = catId;
    document.querySelectorAll('.cat-item').forEach(el => el.classList.remove('active'));
    document.getElementById(`cat-${catId}`).classList.add('active');
    
    // Obtener sugerencias
    google.script.run.withSuccessHandler(msgs => {
      if(msgs && msgs.length > 0) {
        // Seleccionar uno aleatorio de los sugeridos
        const randomMsg = msgs[Math.floor(Math.random() * msgs.length)].message;
        document.getElementById('messageInput').placeholder = "Ejemplo: " + randomMsg;
      }
    }).getMessages(catId);
  }

  function submitRecognition() {
    if(!APP_STATE.selectedPersonId) return showToast("Falta destinatario.");
    if(!APP_STATE.selectedCategoryId) return showToast("Selecciona una categoría.");
    const msg = document.getElementById('messageInput').value.trim();
    if(!msg) return showToast("Escribe un mensaje.");

    const btn = document.getElementById('btnSubmit');
    btn.disabled = true;
    btn.innerText = "Enviando...";

    const payload = {
      personId: APP_STATE.selectedPersonId,
      categoryId: APP_STATE.selectedCategoryId,
      message: msg
    };

    google.script.run
      .withSuccessHandler(res => {
        closeModal();
        showToast(`¡Reconocimiento enviado a ${res.personName}!`);
        // Recargar datos para actualizar podio
        setTimeout(() => location.reload(), 2000); 
      })
      .withFailureHandler(err => {
        showToast(err.message || "Error al enviar");
        btn.disabled = false;
        btn.innerText = "Enviar Reconocimiento 🎉";
      })
      .createRecognition(payload);
  }

  function showToast(msg) {
    const t = document.getElementById('toast');
    t.innerText = msg;
    t.style.display = 'block';
    setTimeout(() => t.style.display = 'none', 4000);
  }

  function handleError(err) {
    document.getElementById('loadingScreen').innerHTML = `<p style="color:red">Error: ${err.message}</p>`;
    console.error(err);
  }

  function escapeHtml(str) {
    if(typeof str !== 'string') return str;
    return str.replace(/[&<>'"]/g, 
      tag => ({
          '&': '&amp;',
          '<': '&lt;',
          '>': '&gt;',
          "'": '&#39;',
          '"': '&quot;'
        }[tag] || tag)
    );
  }
</script>
