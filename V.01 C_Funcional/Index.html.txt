  function showLoadError(error) {

    const message = (error && error.message) || 'Ocurrió un error al cargar los datos.';

    const html = `
      <div class="loading" style="color:var(--maroon);">
        ⚠️ ${escapeHtml(message)}<br>

    document.getElementById('peopleContainer').innerHTML = html;
    document.getElementById('deptContainer').innerHTML = '<div class="dept-empty">No se pudieron cargar los departamentos.</div>';

    showError(error);
  }

  function filterPeople() {

    const query = document.getElementById('searchInput').value.toLowerCase().trim();
    const dept = document.getElementById('departmentFilter').value;

    const filtered = APP.recognizedPeople.filter(person => {

      const matchesQuery =
        !query ||
        String(person.name).toLowerCase().includes(query) ||
        String(person.department).toLowerCase().includes(query);

      const matchesDept = !dept || person.department === dept;

      return matchesQuery && matchesDept;
    });

    renderPeople(filtered);
  }

  function formatTimestamp(value) {

    try {
      const date = new Date(value);
      return date.toLocaleString('es-MX', { day: '2-digit', month: 'short', hour: '2-digit', minute: '2-digit' });
    } catch (e) {
      return '';
    }
  }

  function sendRecognition() {

    if (!APP.selectedPerson) {
      showToast('Selecciona una persona.');
      return;
    }

    if (!APP.selectedCategory) {
      showToast('Selecciona una categoría.');
      return;
    }

    const message = document.getElementById('message').value.trim();

    if (!message) {
      showToast('Escribe un mensaje.');
      return;
    }

    const data = {
      personId: APP.selectedPerson.id,
      categoryId: APP.selectedCategory.id,
      message: message
    };

    const button = document.querySelector('#stepCompose .primary-button');
    button.disabled = true;
    button.textContent = 'Enviando…';

    google.script.run
      .withSuccessHandler(recognitionSent)
      .withFailureHandler(err => {
        button.disabled = false;
        button.textContent = '🎉 Enviar reconocimiento';
        showError(err);
      })
      .createRecognition(data);
  }

      jobTitle: document.getElementById('pf_jobTitle').value,
      photoUrl: document.getElementById('pf_photoUrl').value,
      isAdmin: document.getElementById('pf_isAdmin').checked
    };

    if (APP.editingPersonId) {
      google.script.run
        .withSuccessHandler(() => {
          closeModal('modalPerson');
          showToast('Persona actualizada exitosamente.');
          loadAdminPeople();
          loadApp();
        })
        .withFailureHandler(showError)
        .adminUpdatePerson(APP.editingPersonId, data);
    } else {
      google.script.run
        .withSuccessHandler(() => {
          closeModal('modalPerson');
          showToast('Persona agregada exitosamente.');
          loadAdminPeople();
          loadApp();
        })
        .withFailureHandler(showError)
        .adminAddPerson(data);
    }
  }

  // Previene inyecciones de código y protege variables de strings rompedores (Obligatorio)
  function escapeHtml(unsafe) {
    if (!unsafe) return '';
    return String(unsafe)
         .replace(/&/g, "&amp;")
         .replace(/</g, "&lt;")
         .replace(/>/g, "&gt;")
         .replace(/"/g, "&quot;")
         .replace(/'/g, "&#039;");
  }

  function escapeJs(unsafe) {
    if (!unsafe) return '';
    return String(unsafe).replace(/\\/g, "\\\\").replace(/'/g, "\\'").replace(/"/g, '\\"');
  }

  function openDiagnostics() {
    openModal('modalDiagnostics');
    document.getElementById('diagnosticsContainer').innerHTML = '<div class="loading">Ejecutando diagnóstico…</div>';
    
    google.script.run
      .withSuccessHandler(renderDiagnostics)
      .withFailureHandler(showError)
      .diagnoseSystem();
  }

  function renderDiagnostics(report) {
    const container = document.getElementById('diagnosticsContainer');
    
    if (!report.spreadsheetConnected) {
      container.innerHTML = `
        <div class="loading" style="color:var(--maroon);">
          ❌ No hay conexión con la hoja de cálculo.<br><br>
          ${escapeHtml(report.spreadsheetError)}
        </div>
      `;
      return;
    }

    let html = `
      <div style="font-size:13.5px; line-height:1.6; padding: 10px 0;">
        <div style="margin-bottom:8px;">✅ Conectado a: <strong>${escapeHtml(report.spreadsheetName)}</strong></div>
        <div style="margin-bottom:18px;">Sesión detectada: <strong>${escapeHtml(report.sessionEmail)}</strong> ${report.userRegistered ? '✅ (Cuenta Activa)' : '⚠️ (No encontrado en el sistema)'}</div>
    `;

    report.sheets.forEach(s => {
      let icon = s.exists && s.headersMatch ? '✅' : (s.exists ? '⚠️' : '❌');
      let status = s.exists ? `${s.rowCount} fila(s)` : 'No existe';
      if (s.error) status = s.error;
      if (s.headersMatch === false) status += ' (Encabezados no coinciden)';
      
      html += `<div style="margin-bottom:6px;">${icon} ${escapeHtml(s.name)} &nbsp;&nbsp; <span style="color:var(--ink-soft);">${escapeHtml(status)}</span></div>`;
    });

    html += `</div>`;
    
    container.innerHTML = html;
  }
