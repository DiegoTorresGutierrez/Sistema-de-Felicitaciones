<script>

  let APP = {

    people: [],

    categories: [],

    selectedPerson: null,

    selectedCategory: null

  };


  /**
   * Inicialización
   */

  document.addEventListener(
    'DOMContentLoaded',
    loadApp
  );


  function loadApp() {

    google.script.run

      .withSuccessHandler(
        initializeApp
      )

      .withFailureHandler(
        showError
      )

      .getAppData();
  }


  function initializeApp(data) {

    APP.people =
      data.people || [];

    APP.categories =
      data.categories || [];

    renderStats(
      data.stats
    );

    renderPeople(
      APP.people
    );
  }


  /**
   * Estadísticas
   */

  function renderStats(stats) {

    document
      .getElementById(
        'totalRecognitions'
      )
      .textContent =
      stats.total || 0;


    document
      .getElementById(
        'monthlyRecognitions'
      )
      .textContent =
      stats.thisMonth || 0;


    document
      .getElementById(
        'peopleRecognized'
      )
      .textContent =
      stats.peopleRecognized || 0;
  }


  /**
   * Personas
   */

  function renderPeople(people) {

    const container =
      document.getElementById(
        'peopleContainer'
      );

    if (!people.length) {

      container.innerHTML =
        '<div class="loading">No se encontraron personas.</div>';

      return;
    }


    container.innerHTML =
      people.map(person => `

        <div
          class="person-card"
          onclick="openRecognition(${escapeJs(person.id)})">

          <img
            class="person-photo"
            src="${escapeHtml(person.photoUrl)}"
            onerror="this.src='${escapeHtml('https://via.placeholder.com/300x300.png?text=Employee')}'">

          <div class="person-name">
            ${escapeHtml(person.name)}
          </div>

          <div class="person-department">
            ${escapeHtml(person.department)}
          </div>

          <div class="person-count">
            🏆 ${person.recognitions}
            reconocimientos
          </div>

        </div>

      `).join('');
  }


  /**
   * Buscar
   */

  function filterPeople() {

    const query =
      document
        .getElementById(
          'searchInput'
        )
        .value
        .toLowerCase()
        .trim();


    const filtered =
      APP.people.filter(person =>

        String(person.name)
          .toLowerCase()
          .includes(query)

        ||

        String(person.department)
          .toLowerCase()
          .includes(query)

      );


    renderPeople(filtered);
  }


  /**
   * Abrir reconocimiento
   */

  function openRecognition(id) {

    const person =
      APP.people.find(
        p =>
          String(p.id) ===
          String(id)
      );


    if (!person) return;


    APP.selectedPerson =
      person;

    APP.selectedCategory =
      null;


    document
      .getElementById(
        'selectedPerson'
      )
      .innerHTML = `

        <img
          src="${escapeHtml(person.photoUrl)}"
          onerror="this.src='https://via.placeholder.com/300x300.png?text=Employee'">

        <h3>
          ${escapeHtml(person.name)}
        </h3>

        <p>
          ${escapeHtml(person.department)}
        </p>

      `;


    renderCategories();


    document
      .getElementById(
        'message'
      )
      .value = '';


    document
      .getElementById(
        'modal'
      )
      .style.display =
      'flex';
  }


  /**
   * Categorías
   */

  function renderCategories() {

    const container =
      document.getElementById(
        'categoriesContainer'
      );


    container.innerHTML =
      APP.categories.map(category => `

        <div
          class="category"
          data-id="${escapeHtml(category.id)}"
          onclick="selectCategory('${escapeJs(category.id)}')">

          <span class="category-emoji">
            ${escapeHtml(category.emoji)}
          </span>

          <span class="category-name">
            ${escapeHtml(category.name)}
          </span>

        </div>

      `).join('');
  }


  function selectCategory(id) {

    APP.selectedCategory =
      APP.categories.find(
        c =>
          String(c.id) ===
          String(id)
      );


    document
      .querySelectorAll(
        '.category'
      )
      .forEach(el => {

        el.classList.toggle(
          'selected',
          el.dataset.id === String(id)
        );

      });


    loadMessages(id);
  }


  /**
   * Mensajes sugeridos
   */

  function loadMessages(categoryId) {

    google.script.run

      .withSuccessHandler(messages => {

        if (!messages.length) return;

        const message =
          document.getElementById(
            'message'
          );

        message.placeholder =
          messages[0].message;

      })

      .getMessages(categoryId);
  }


  /**
   * Enviar
   */

  function sendRecognition() {

    if (!APP.selectedPerson) {

      showToast(
        'Selecciona una persona.'
      );

      return;
    }


    if (!APP.selectedCategory) {

      showToast(
        'Selecciona una categoría.'
      );

      return;
    }


    const message =
      document
        .getElementById(
          'message'
        )
        .value
        .trim();


    if (!message) {

      showToast(
        'Escribe un mensaje.'
      );

      return;
    }


    const data = {

      personId:
        APP.selectedPerson.id,

      categoryId:
        APP.selectedCategory.id,

      message:
        message

    };


    google.script.run

      .withSuccessHandler(
        recognitionSent
      )

      .withFailureHandler(
        showError
      )

      .createRecognition(data);
  }


  /**
   * Reconocimiento enviado
   */

  function recognitionSent(result) {

    closeModal();

    showToast(
      `🎉 ¡Reconocimiento enviado a ${result.personName}!`
    );


    setTimeout(
      loadApp,
      1000
    );
  }


  /**
   * Cerrar modal
   */

  function closeModal() {

    document
      .getElementById(
        'modal'
      )
      .style.display =
      'none';
  }


  /**
   * Mensajes UI
   */

  function showToast(message) {

    const toast =
      document.getElementById(
        'toast'
      );

    toast.textContent =
      message;

    toast.style.display =
      'block';


    setTimeout(() => {

      toast.style.display =
        'none';

    }, 3500);
  }


  function showError(error) {

    console.error(error);

    showToast(
      error.message ||
      'Ocurrió un error.'
    );
  }


  /**
   * Protección básica para HTML.
   */

  function escapeHtml(value) {

    return String(value ?? '')
      .replace(/&/g, '&amp;')
      .replace(/</g, '&lt;')
      .replace(/>/g, '&gt;')
      .replace(/"/g, '&quot;')
      .replace(/'/g, '&#039;');
  }


  function escapeJs(value) {

    return String(value ?? '')
      .replace(/\\/g, '\\\\')
      .replace(/'/g, "\\'");
  }
